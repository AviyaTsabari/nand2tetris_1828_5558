VMT
librec_info: #(path: #(), lib_committed: "")
class
	{
	Main(inputPath)
		{
		cw = .createCodeWriter(inputPath)
		cw.WriteInit()

		if .isVmFile(inputPath)
			.translateSingleFile(inputPath, cw)
		else
			.translateDirectory(inputPath, cw)

		cw.Close()
		}

	createCodeWriter(inputPath)
		{
		if .isVmFile(inputPath)
			outputFile = inputPath.BeforeLast(".") $ ".asm"
		else
			outputFile = inputPath $ "/" $ .getDirectoryName(inputPath) $ ".asm"

		out = File(outputFile, "w")
		cw = CodeWriter(out)

		return cw
		}

	getDirectoryName(path)
		{
		dirName = path.AfterLast("/")
		if dirName is ""
		   dirName = path.AfterLast("\\")
		return dirName
		}

	isVmFile(path)
		{
		return path.Suffix?(".vm")
		}

	translateDirectory(dir, cw)
		{
		for file in Dir(dir $ "/*.vm")
			.translateSingleFile(dir $ "/" $ file, cw)
		}

	translateSingleFile(inputFile, cw)
		{
		cw.SetFileName(.getFileName(inputFile))

		parser = Parser(inputFile)

		while true
			{
			.translateCommand(parser, cw)

			if parser.HasMoreCommands()
				parser.Advance()
			else
				break
			}
		}

	getFileName(inputFile)
		{
		fileName = inputFile.AfterLast("/")
		if fileName is ""
		   fileName = inputFile.AfterLast("\\")
		return fileName.BeforeLast(".")
		}

	translateCommand(parser, cw)
		{
		type = parser.CommandType()

		if type is Parser.C_ARITHMETIC
			cw.WriteArithmetic(parser.Arg1())
		else if type is Parser.C_PUSH or type is Parser.C_POP
			cw.WritePushPop(type, parser.Arg1(), parser.Arg2())
		else if type is Parser.C_LABEL
			cw.WriteLabel(parser.Arg1())
		else if type is Parser.C_GOTO
			cw.WriteGoto(parser.Arg1())
		else if type is Parser.C_IF
			cw.WriteIf(parser.Arg1())
		else if type is Parser.C_FUNCTION
			cw.WriteFunction(parser.Arg1(), parser.Arg2())
		else if type is Parser.C_CALL
			cw.WriteCall(parser.Arg1(), parser.Arg2())
		else if type is Parser.C_RETURN
			cw.WriteReturn()
		else
			throw "Unsupported command: " $ parser.CurrentInstruction()
		}
	}
