		while .pos < .text.Size()
			.readNextToken()

		return .tokens
		}

	readNextToken()
		{
		ch = .text[.pos]

		if .handleWhiteSpace(ch)
			return

		if .handleComment(ch)
			return

		if .handleInteger(ch)
			return

		if .handleString(ch)
			return

		if .handleSymbol(ch)
			return

		if .handleIdentifier(ch)
			return

		.pos++
		}

	handleWhiteSpace(ch)
		{
		if not .isWhiteSpace(ch)
			return false

		.pos++
		return true
		}

	handleComment(ch)
		{
		if ch isnt "/" or not .hasNext()
			return false

		next = .text[.pos + 1]

		if next is "/"
			{
			.skipLineComment()
			return true
			}

		if next is "*"
			{
			.skipBlockComment()
			return true
			}

		return false
		}

	handleInteger(ch)
		{
		if not .isDigit(ch)
			return false

		.tokens.Add(Object(type: "integerConstant", value: .readInteger()))
		return true
		}

	handleString(ch)
		{
		if ch isnt '"'
			return false

		.tokens.Add(Object(type: "stringConstant", value: .readString()))
		return true
		}

	handleSymbol(ch)
		{
		if not .isSymbol(ch)
			return false

		.tokens.Add(Object(type: "symbol", value: ch))
		.pos++
		return true
		}

	handleIdentifier(ch)
		{
		if not .isLetterOrUnderscore(ch)
			return false

		word = .readIdentifier()
		type = .isKeyword(word) ? "keyword" : "identifier"
		.tokens.Add(Object(type: type, value: word))
		return true
		}

		hasNext()
		{
		return .pos + 1 < .text.Size()
		}

	skipLineComment()
		{
		.pos += 2

		while .pos < .text.Size() and .text[.pos] isnt "\n"
			.pos++
		}

	skipBlockComment()
		{
		.pos += 2

		while .pos + 1 < .text.Size()
			{
			if .text[.pos] is "*" and .text[.pos + 1] is "/"
				{
				.pos += 2
				return
				}

			.pos++
			}
		}

	readInteger()
		{
		start = .pos

		while .pos < .text.Size() and .isDigit(.text[.pos])
			.pos++

		return .text[start :: .pos - start]
		}

			.pos++

		value = .text[start :: .pos - start]

		.pos++ // ?????? ?? ??????? ???????

		return value
		}

	readIdentifier()
		{
		start = .pos

		while .pos < .text.Size() and .isIdentifierChar(.text[.pos])
			.pos++

		return .text[start :: .pos - start]
		}

	isWhiteSpace(ch)
		{
		return ch is " " or ch is "\t" or ch is "\r" or ch is "\n"
		}

	isDigit(ch)
		{
		return ch >= "0" and ch <= "9"
		}

	isLetterOrUnderscore(ch)
		{
		return (ch >= "a" and ch <= "z") or

		{
		return .isLetterOrUnderscore(ch) or .isDigit(ch)
		}

	isSymbol(ch)
		{
		return .symbols.Has?(ch)
		}

	isKeyword(word)
		{
		return .keywords.Has?(word)
		}

		{
		type = token.type
		value = .xmlValue(token.value)

		return "<" $ type $ "> " $
			value $
			" </" $ type $ ">"
		}

		{
		lines = Object()

		lines.Add("<tokens>")

		for token in .tokens
			lines.Add(.tokenToXml(token))

		return lines
		}

	xmlValue(value)
		{
		if value is "<"
			return "&lt;"

		if value is ">"
			return "&gt;"

		if value is "&"
			return "&amp;"

		return value
		}

	WriteXml(outputFile)
		{
		lines = .ToXmlLines()
		out = File(outputFile, "w")

		for line in lines
			out.Write(line $ "\n")

		out.Close()
		}

class
	{
	New(inputText)
		{
		.text = inputText
		.pos = 0
		.tokens = Object()
		.currentIndex = -1
		.currentToken = false

		.keywords = Object(
			"class", "constructor", "function", "method",
			"field", "static", "var",
			"int", "char", "boolean", "void",
			"true", "false", "null", "this",
			"let", "do", "if", "else", "while", "return")
		.symbols = "{}()[].,;+-*/&|<>=~"
		}

	Tokenize()
		{
		while .pos < .text.Size()
			.readNextToken()

		.currentIndex = -1
		.currentToken = false

		return .tokens
		}

	HasMoreTokens()
		{
		return .currentIndex + 1 < .tokens.Size()
		}

	Advance()
		{
		if not .HasMoreTokens()
			throw "No more tokens"

		.currentIndex++
		.currentToken = .tokens[.currentIndex]
		}

	TokenType()
		{
		return .currentToken.type
		}

	Keyword()
		{
		return .currentToken.value
		}

	Symbol()
		{
		return .currentToken.value
		}

	Identifier()
		{
		return .currentToken.value
		}

	IntVal()
		{
		return Number(.currentToken.value)
		}

	StringVal()
		{
		return .currentToken.value
		}

	readNextToken()
		{
		ch = .text[.pos]

		if .handleWhiteSpace(ch)
			return

		if .handleComment(ch)
			return

		if .handleInteger(ch)
			return

		if .handleString(ch)
			return

		if .handleSymbol(ch)
			return

		}

		.pos++
		return true
		}

	handleComment(ch)
		{

			return false

		next = .text[.pos + 1]

		if next is "/"
			{
			.skipLineComment()
			return true
			}

		if next is "*"
			{
			.skipBlockComment()
			return true
			}

		return false
		}

		return true

		.tokens.Add(Object(type: "symbol", value: ch))
		.pos++
		return true
		}

			return false

		word = .readIdentifier()
		type = .isKeyword(word) ? "keyword" : "identifier"
		.tokens.Add(Object(type: type, value: word))
		return true
		}

	hasNext()
		{
		return .pos + 1 < .text.Size()
		}

	skipLineComment()
		{
		.pos += 2

		while .pos < .text.Size() and .text[.pos] isnt "\n"
			.pos++
		}

	skipBlockComment()
		{
		.pos += 2

		while .pos + 1 < .text.Size()

			if .text[.pos] is "*" and .text[.pos + 1] is "/"

				.pos += 2
				return
				}

			.pos++
			}

		start = .pos

		while .pos < .text.Size() and .isDigit(.text[.pos])

		return .text[start :: .pos - start]

		while .pos < .text.Size() and .text[.pos] isnt '"'

		value = .text[start :: .pos - start]

		.pos++

		return value
